cmake_minimum_required(VERSION 3.20)
project(nixdatifier-tray LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets Svg)
qt_add_executable(nixdatifier-tray main.cpp TrayIconRenderer.cpp)
target_link_libraries(nixdatifier-tray PRIVATE Qt6::Widgets Qt6::Svg)
install(TARGETS nixdatifier-tray RUNTIME DESTINATION bin)

option(NIXDATIFIER_TRAY_TESTS "Build the tray renderer checks" OFF)
if(NIXDATIFIER_TRAY_TESTS)
    qt_add_executable(tray-renderer-test tests/renderer.cpp TrayIconRenderer.cpp)
    target_include_directories(tray-renderer-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(tray-renderer-test PRIVATE Qt6::Widgets Qt6::Svg)
endif()
